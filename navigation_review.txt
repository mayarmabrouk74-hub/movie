  @override
  void initState() {
    super.initState();

    Timer(const Duration(seconds: 3), () {
      if (!mounted) return;

import 'dart:async';
import 'package:flutter/material.dart';

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {

  static const Color yellow = Color(0xFFF6BD00);
  static const Color bg = Color(0xFF121312);

  final PageController _controller = PageController(
    viewportFraction: 0.62,
    initialPage: 1000,
  );

  Timer? _timer;

  // MAIN CAROUSEL MOVIES

  final List<String> movies = [
    'assets/movie1.jpg',
    'assets/onboarding6.png',
    'assets/movie2.jpg',
  ];

  // ACTION MOVIES

  final List<String> actionMovies = [
    'assets/movie2.jpg',
    'assets/movie3.jpg',
    'assets/movie4.jpg',
  ];

  @override
  void initState() {
    super.initState();

    // AUTOMATIC CAROUSEL

    _timer = Timer.periodic(
      const Duration(seconds: 3),
      (timer) {
        if (_controller.hasClients) {
          _controller.nextPage(
            duration: const Duration(milliseconds: 700),
            curve: Curves.easeInOut,
          );
        }
      },
    );
  }

  @override
  void dispose() {
    _timer?.cancel();
    _controller.dispose();
    super.dispose();
  }

  // ==========================================
  // RATING BADGE
  // ==========================================

  Widget ratingBadge(double screenWidth) {
    return Container(
      padding: EdgeInsets.symmetric(
        horizontal: screenWidth * 0.025,
        vertical: screenWidth * 0.012,
      ),

      decoration: BoxDecoration(
        color: const Color(0xCC282A28),
        borderRadius: BorderRadius.circular(30),
      ),

      child: Row(
        mainAxisSize: MainAxisSize.min,
        children: [

          Text(
            '7.7',
            style: TextStyle(
              color: Colors.white,
              fontSize: (screenWidth * 0.035)
                  .clamp(11.0, 17.0),
              fontWeight: FontWeight.w500,
            ),
          ),

          SizedBox(
            width: screenWidth * 0.012,
          ),

          Icon(
            Icons.star,
            color: yellow,
            size: (screenWidth * 0.045)
                .clamp(14.0, 22.0),
          ),
        ],
      ),
    );
  }

  // ==========================================
  // MAIN MOVIE CARD
  // ==========================================

  Widget mainMovieCard({
    required String image,
    required double width,
    required double height,
    required double screenWidth,
  }) {
    return SizedBox(
      width: width,
      height: height,

      child: ClipRRect(
        borderRadius: BorderRadius.circular(
          screenWidth * 0.045,
        ),

        child: Stack(
          fit: StackFit.expand,

          children: [

            Image.asset(
              image,
              fit: BoxFit.cover,
            ),

            // RATING

            Positioned(
              top: screenWidth * 0.025,
              left: screenWidth * 0.025,

              child: ratingBadge(screenWidth),
            ),
          ],
        ),
      ),
    );
  }

  // ==========================================
  // ACTION MOVIE CARD
  // ==========================================

  Widget actionMovieCard({
    required String image,
    required double width,
    required double height,
    required double screenWidth,
  }) {
    return Container(
      width: width,
      height: height,

      margin: EdgeInsets.only(
        right: screenWidth * 0.035,
      ),

      child: ClipRRect(
        borderRadius: BorderRadius.circular(
          screenWidth * 0.045,
        ),

        child: Stack(

  // ==========================================
  // BUILD
  // ==========================================

  @override
  Widget build(BuildContext context) {

    final screenSize = MediaQuery.sizeOf(context);

    final screenWidth = screenSize.width;
    final screenHeight = screenSize.height;

    // RESPONSIVE SIZES
